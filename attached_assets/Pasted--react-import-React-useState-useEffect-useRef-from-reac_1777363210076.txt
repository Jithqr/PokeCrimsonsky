export default function App() {
  const [scene, setScene] = useState('intro1');
  const [playerName, setPlayerName] = useState('');
  const [playerStarter, setPlayerStarter] = useState('');
  const [mentorPath, setMentorPath] = useState(''); // 'oak' or 'gio'
  
  const [displayedText, setDisplayedText] = useState('');
  const [isTyping, setIsTyping] = useState(false);
  const [showNameModal, setShowNameModal] = useState(false);
  const [tempName, setTempName] = useState('');

  const intervalRef = useRef(null);

  const getFullText = () => {
    let text = STORY[scene].text;
    text = text.replace(/{name}/g, playerName);
    text = text.replace(/{starter}/g, playerStarter);
    return text;
  };

  const startTyping = () => {
    const fullText = getFullText();
    setDisplayedText('');
    setIsTyping(true);
    let i = 0;

    if (intervalRef.current) clearInterval(intervalRef.current);

    intervalRef.current = setInterval(() => {
      setDisplayedText(fullText.slice(0, i + 1));
      i++;
      if (i >= fullText.length) {
        clearInterval(intervalRef.current);
        setIsTyping(false);
      }
    }, 25); // Typing speed
  };

  useEffect(() => {
    startTyping();
    return () => {
      if (intervalRef.current) clearInterval(intervalRef.current);
    };
  }, [scene]);

  const handleSkipTyping = () => {
    if (isTyping) {
      if (intervalRef.current) clearInterval(intervalRef.current);
      setDisplayedText(getFullText());
      setIsTyping(false);
    }
  };

  const handleChoice = (choice) => {
    if (choice.action === 'ASK_NAME') {
      setMentorPath(choice.path);
      setShowNameModal(true);
    } else if (choice.action === 'CHOOSE_STARTER') {
      setPlayerStarter(choice.starter);
      setScene('epilogue');
    } else if (choice.action === 'RESTART') {
      setPlayerName('');
      setPlayerStarter('');
      setMentorPath('');
      setScene('intro1');
    } else if (choice.next) {
      setScene(choice.next);
    }
  };

  const handleNameSubmit = () => {
    if (tempName.trim()) {
      setPlayerName(tempName.trim());
      setShowNameModal(false);
      setTempName('');
      // Advance to the explanation scene based on chosen path
      setScene(mentorPath === 'oak' ? 'explain_oak' : 'explain_gio');
    }
  };

  const currentSceneData = STORY[scene];

  return (
    <div className="min-h-screen bg-black text-neutral-300 font-sans flex flex-col items-center justify-center p-4 selection:bg-neutral-800">
      
      {/* Main Game Container */}
      <div className="max-w-2xl w-full bg-neutral-950 border border-neutral-800 rounded-2xl shadow-2xl p-6 md:p-10 relative overflow-hidden transition-all duration-300">
        
        {/* Decorative Top Accent */}
        <div className="absolute top-0 left-0 w-full h-1 bg-gradient-to-r from-neutral-900 via-neutral-500 to-neutral-900 opacity-30"></div>

        {/* Scene Image Box */}
        {currentSceneData.image && (
          <div className="w-full h-48 md:h-56 mb-6 rounded-xl overflow-hidden border border-neutral-800 bg-neutral-900 relative">
            <img 
              src={currentSceneData.image} 
              alt="Scene visual" 
              className="w-full h-full object-cover opacity-70 mix-blend-luminosity hover:mix-blend-normal hover:opacity-100 transition-all duration-700"
            />
            <div className="absolute inset-0 bg-gradient-to-t from-neutral-950 via-transparent to-transparent pointer-events-none"></div>
          </div>
        )}

        {/* Text Display Area */}
        <div 
          className="min-h-[120px] mb-8 cursor-pointer group relative"
          onClick={handleSkipTyping}
        >
          <p className="text-lg md:text-xl leading-relaxed whitespace-pre-wrap text-neutral-200 drop-shadow-sm">
            {displayedText}
            {isTyping && <span className="animate-pulse inline-block w-2 h-5 bg-neutral-400 ml-1 translate-y-1"></span>}
          </p>
          
          {/* Skip typing indicator */}
          {isTyping && (
            <div className="absolute -bottom-6 right-0 flex items-center gap-1 text-xs text-neutral-600 opacity-0 group-hover:opacity-100 transition-opacity">
              <FastForward className="w-3 h-3" />
              <span>Click to skip</span>
            </div>
          )}
        </div>

        {/* Action Choices */}
        <div className="min-h-[160px] flex flex-col justify-end">
          {!isTyping && (
            <div className="flex flex-col gap-3 animate-in fade-in slide-in-from-bottom-4 duration-700">
              {currentSceneData.choices.map((choice, idx) => (
                <button
                  key={idx}
                  onClick={() => handleChoice(choice)}
                  className="flex items-center justify-between w-full text-left bg-neutral-900 hover:bg-neutral-800 border border-neutral-800 hover:border-neutral-500 px-6 py-4 rounded-xl transition-all duration-300 group shadow-sm hover:shadow-md"
                >
                  <span className="text-neutral-300 group-hover:text-white font-medium text-base tracking-wide">
                    {choice.label}
                  </span>
                  <ChevronRight className="w-5 h-5 text-neutral-600 group-hover:text-neutral-300 transition-colors transform group-hover:translate-x-1 duration-300" />
                </button>
              ))}
            </div>
          )}
        </div>
      </div>

      {/* Custom Name Input Modal */}
      {showNameModal && (
        <div className="fixed inset-0 bg-black/90 flex items-center justify-center z-50 px-4 backdrop-blur-sm">
          <div className="bg-neutral-950 border border-neutral-700 p-8 rounded-2xl w-full max-w-sm shadow-2xl animate-in zoom-in-95 duration-200">
            <div className="flex items-center gap-3 mb-6">
              <div className="p-2 bg-neutral-900 rounded-full border border-neutral-800">
                <User className="w-5 h-5 text-neutral-400" />
              </div>
              <h2 className="text-xl font-medium text-neutral-100">Enter your name</h2>
            </div>
            
            <input
              type="text"
              value={tempName}
              onChange={(e) => setTempName(e.target.value)}
              onKeyDown={(e) => e.key === 'Enter' && tempName.trim() && handleNameSubmit()}
              className="w-full bg-black border border-neutral-800 text-neutral-200 px-4 py-3 rounded-xl mb-8 focus:outline-none focus:border-neutral-500 focus:ring-1 focus:ring-neutral-500 transition-all text-lg"
              placeholder="Your name..."
              autoFocus
              maxLength={15}
            />
            
            <button
              onClick={handleNameSubmit}
              disabled={!tempName.trim()}
              className="w-full bg-neutral-200 text-black font-semibold py-3.5 rounded-xl hover:bg-white transition-all disabled:opacity-30 disabled:cursor-not-allowed shadow-[0_0_15px_rgba(255,255,255,0.1)] hover:shadow-[0_0_20px_rgba(255,255,255,0.2)]"
            >
              Confirm
            </button>
          </div>
        </div>
      )}
    </div>
  );
}
