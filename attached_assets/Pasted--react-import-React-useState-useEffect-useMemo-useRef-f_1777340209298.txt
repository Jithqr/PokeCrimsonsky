// --- API Cache to prevent redundant requests ---
const apiCache = {};
const fetchWithCache = async (url) => {
  if (apiCache[url]) return apiCache[url];
  try {
    const res = await fetch(url);
    const data = await res.json();
    apiCache[url] = data;
    return data;
  } catch (err) {
    console.error(`Failed to fetch ${url}`, err);
    return null;
  }
};

const calculateStat = (base, iv, ev, level, nature, statName) => {
  if (statName === 'hp') {
    return Math.floor(((2 * base + iv + Math.floor(ev / 4)) * level) / 100) + level + 10;
  }
  let natureMod = 1;
  if (NATURES[nature]?.up === statName) natureMod = 1.1;
  if (NATURES[nature]?.down === statName) natureMod = 0.9;
  
  const rawStat = Math.floor(((2 * base + iv + Math.floor(ev / 4)) * level) / 100) + 5;
  return Math.floor(rawStat * natureMod);
};

const AbilityCard = ({ abilityInfo }) => {
  const [expanded, setExpanded] = useState(false);
  const [details, setDetails] = useState(null);

  useEffect(() => {
    // Pre-fetch ability data so quick summary is available immediately if we want it,
    // but here we match the design which shows the summary even when collapsed (or we expand it by default)
    fetchWithCache(abilityInfo.ability.url).then(setDetails);
  }, [abilityInfo.ability.url]);

  const getEnglishEntry = (entries, key = 'flavor_text') => {
    if (!entries) return 'Loading...';
    const entry = entries.find(e => e.language.name === 'en');
    return entry ? entry[key].replace(/\f/g, ' ') : 'No description available.';
  };

  return (
    <div className="bg-[#121212] border border-zinc-800 rounded-xl mb-3 overflow-hidden transition-all">
      <button 
        onClick={() => setExpanded(!expanded)}
        className="w-full flex items-start justify-between p-4 focus:outline-none text-left"
      >
        <div className="pr-4">
          <div className="flex items-center gap-2 mb-1.5">
            <span className="text-zinc-100 font-semibold capitalize text-base">
              {abilityInfo.ability.name.replace('-', ' ')}
            </span>
            {abilityInfo.is_hidden && (
              <span className="bg-red-500/90 text-white text-[10px] px-1.5 py-0.5 rounded uppercase tracking-wide font-bold">
                Hidden
              </span>
            )}
          </div>
          <p className="text-zinc-400 text-sm leading-snug line-clamp-2">
            {details ? getEnglishEntry(details.flavor_text_entries) : 'Loading ability info...'}
          </p>
        </div>
        <div className="mt-1">
          {expanded ? <ChevronUp size={18} className="text-zinc-500" /> : <ChevronDown size={18} className="text-zinc-500" />}
        </div>
      </button>

      {expanded && details && (
        <div className="px-4 pb-4 border-t border-zinc-800/50 pt-3">
          <h4 className="text-red-500 text-xs uppercase font-semibold mb-1 tracking-wider">Detailed Effect</h4>
          <p className="text-zinc-300 text-sm leading-relaxed">
            {getEnglishEntry(details.effect_entries, 'effect')}
          </p>
        </div>
      )}
    </div>
  );
};
